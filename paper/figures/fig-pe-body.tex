%% The target design is drawn, not the prototype RTL (see the figure caption).
%% Coordinates are in bp.  Needs: \usepackage{scalenative-figstyle}.
\begin{AFcanvas}{F}{240}
  % Local shorthands.  Each is a composition of keys of scalenative-figstyle.sty -- no colour, dash pattern, width or
  % size of its own:
  %   pe open bi   a connection that the design has not fixed yet
  %   pe err wire  a branch of the error line that joins the trunk (the trunk carries the tip)
  %   pe port wire the short bus-level stub between a port on the PE boundary and the NoC endpoint
  \tikzset{
    pe open bi/.style={af data bi, dash pattern=on 1.2bp off 1.2bp, draw=AFSlotS},
    pe err wire/.style={af err, -},
    pe port wire/.style={af bus, -},
  }
  % Rows of the hand-off path (y of outlet 0 / outlet 1) and the level half-way between them
  \def\peYa{112bp}\def\peYb{66bp}\def\peYm{89bp}
  % Tracks of the top band (control bus below, error trunk above) and of the right-hand spine
  \def\peYbus{157bp}\def\peYerr{165bp}
  \def\peXl{478.5bp}\def\peXc{487.5bp}\def\peXe{496.5bp}
  % ================= PE boundary =================
  \AFboxoutermost[af unit]{pe}{(12bp,1.2bp)}{(503.2bp,171.6bp)}
  \node[af frame label, anchor=north west] (pel) at (15bp,169.6bp) {PE};
  % ================= control column =================
  \node[af blk=Cmd, anchor=north west] (cp) at (16.5bp,153bp) {Command\\processor};

  \node[af lbl, anchor=south east] (l1l) at ($(l1i.north east)+(0,0.5bp)$) {L1};
  \AFfit[af group=Lone, inner sep=2.2bp]{l1}{(l1d)(l1i)(l1l)}
  \node[af blk dark=DMA, anchor=south west] (dma) at (16.5bp,38bp) {DMA};
  \node[af blk=NoC, anchor=south west, minimum width=40bp, minimum height=22bp] (ep) at (23.4bp,4.5bp) {NoC\\endpoint};   % left edge = left edge of the L1 frame
  \node[af port=east] (pin) at (10.44bp,20.5bp) {};
  \node[af port=west] (pout) at (13.56bp,10.5bp) {};

  \node[af blk=Ltwo, anchor=south west, minimum width=416.6bp, minimum height=10.5bp, inner ysep=0pt] (arb) at (78.9bp,16bp) {L2 arbiter};
  % ================= Tensor unit =================
